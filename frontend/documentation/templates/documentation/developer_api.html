{% trans "Amounts and precision" %}

{% blocktrans trimmed %}All money fields use whole numbers in satoshis. One TAN is one hundred million satoshis. Do not use float types for money in your code.{% endblocktrans %}

tan = satoshis / 100_000_000

{% trans "Read-only endpoints" %}

  • GET /api/chain/status/ — {% trans "height, tip, mempool count, reference price, difficulty" %}
  • GET /api/chain/blocks/?limit=N — {% trans "recent blocks as JSON" %}
  • GET /api/chain/block/<hash>/ — {% trans "one block with transactions" %}
  • GET /api/chain/tx/<txid>/ — {% trans "transaction plus confirmed flag" %}
  • GET /api/chain/utxo/<address>/ — {% trans "spendable outputs for an address" %}
  • GET /api/chain/mempool/ — {% trans "pending transactions" %}
  • GET /api/contracts/ — {% trans "deployed contracts list" %}

{% trans "Write endpoints (POST)" %}

{% blocktrans trimmed %}Submitting transactions and mining changes local data. In production, add authentication, rate limits, and network rules.{% endblocktrans %}

  • POST /api/chain/tx/submit/ — {% trans "body includes transaction object" %}
  • POST /api/chain/mine/ — {% trans "optional miner_wif for who receives fees" %}
  • POST /api/chain/price/ — {% trans "set reference_price_usd" %}
  • POST /api/contracts/deploy/ — {% trans "deployer_wif, fee_satoshis, bytecode_hex" %}
  • POST /api/contracts/call/ — {% trans "caller_wif, contract_id_hex, calldata_hex, fee_satoshis" %}
{% blocktrans trimmed %}POST routes use csrf_exempt for simple JSON tools. Real browser apps should send CSRF tokens or hide writes behind your own backend.{% endblocktrans %}
